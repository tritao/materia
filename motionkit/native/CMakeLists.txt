cmake_minimum_required(VERSION 3.21)
project(motionkit_core VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
if(NOT DEFINED RK_BUILD_SHARED)
    set(RK_BUILD_SHARED ON)
endif()
option(MK_BUILD_SHARED "Build MotionKit as a shared library" ${RK_BUILD_SHARED})

# Only Ruckig's offline Community calculator is linked. Its upstream defaults
# enable the networked cloud client, examples and tests, so override them here.
set(BUILD_CLOUD_CLIENT OFF)
set(BUILD_EXAMPLES OFF)
set(BUILD_PYTHON_MODULE OFF)
set(BUILD_TESTS OFF)
set(BUILD_BENCHMARK OFF)
set(BUILD_SHARED_LIBS OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ruckig/CMakeLists.txt")
    message(FATAL_ERROR "Initialize motionkit/native/vendor/ruckig submodule")
endif()
add_subdirectory(vendor/ruckig EXCLUDE_FROM_ALL)
set_target_properties(ruckig PROPERTIES POSITION_INDEPENDENT_CODE ON)
get_target_property(MK_RUCKIG_SOURCES ruckig SOURCES)
foreach(MK_RUCKIG_SOURCE IN LISTS MK_RUCKIG_SOURCES)
    if(MK_RUCKIG_SOURCE MATCHES "cloud_client")
        message(FATAL_ERROR "MotionKit must not compile Ruckig's cloud client")
    endif()
endforeach()

# TOPP-RA's Seidel solver is the only optimizer compiled for path timing.
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/toppra/cpp/CMakeLists.txt")
    message(FATAL_ERROR "Initialize motionkit/native/vendor/toppra submodule")
endif()
set(MK_OWN_BUILD_TESTING ${BUILD_TESTING})
set(BUILD_TESTING OFF)
set(PYTHON_BINDINGS OFF)
set(BUILD_WITH_PINOCCHIO OFF)
set(BUILD_WITH_qpOASES OFF)
set(BUILD_WITH_GLPK OFF)
add_subdirectory(vendor/toppra/cpp EXCLUDE_FROM_ALL)
set(BUILD_TESTING ${MK_OWN_BUILD_TESTING})
get_target_property(MK_TOPPRA_SOURCES toppra SOURCES)
list(FILTER MK_TOPPRA_SOURCES EXCLUDE REGEX "(qpOASES|glpk|joint_torque|cartesian_velocity_norm|parametrizer)")
set_property(TARGET toppra PROPERTY SOURCES "${MK_TOPPRA_SOURCES}")
foreach(MK_TOPPRA_SOURCE IN LISTS MK_TOPPRA_SOURCES)
    if(MK_TOPPRA_SOURCE MATCHES "(qpOASES|glpk)")
        message(FATAL_ERROR "MotionKit must not compile TOPP-RA optional solvers")
    endif()
endforeach()

if(MK_BUILD_SHARED)
    set(MK_LIBRARY_TYPE SHARED)
else()
    set(MK_LIBRARY_TYPE STATIC)
endif()

add_library(motionkit_core ${MK_LIBRARY_TYPE} src/trajectory.cpp src/validation.cpp src/generator.cpp src/path.cpp)
target_compile_features(motionkit_core PUBLIC cxx_std_17)
target_link_libraries(motionkit_core PRIVATE ruckig::ruckig toppra)
target_include_directories(motionkit_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
target_compile_definitions(motionkit_core PRIVATE MK_BUILDING_LIBRARY)
if(NOT MK_BUILD_SHARED)
    target_compile_definitions(motionkit_core PUBLIC MK_STATIC)
endif()
set_target_properties(motionkit_core PROPERTIES
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES
    VERSION ${PROJECT_VERSION}
    SOVERSION 0)

if(BUILD_TESTING)
    add_executable(motionkit_trajectory_tests tests/trajectory.cpp)
    target_link_libraries(motionkit_trajectory_tests PRIVATE motionkit_core)
    add_test(NAME motionkit_trajectory COMMAND motionkit_trajectory_tests)
    add_executable(motionkit_validation_tests tests/validation.cpp)
    target_link_libraries(motionkit_validation_tests PRIVATE motionkit_core)
    add_test(NAME motionkit_validation COMMAND motionkit_validation_tests)
    add_executable(motionkit_generator_tests tests/generator.cpp)
    target_link_libraries(motionkit_generator_tests PRIVATE motionkit_core ruckig::ruckig)
    add_test(NAME motionkit_generator COMMAND motionkit_generator_tests)
    add_executable(motionkit_path_tests tests/path.cpp)
    target_link_libraries(motionkit_path_tests PRIVATE motionkit_core)
    add_test(NAME motionkit_path COMMAND motionkit_path_tests)
endif()
