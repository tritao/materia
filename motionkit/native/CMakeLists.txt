cmake_minimum_required(VERSION 3.21)
project(motionkit_core VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
if(NOT DEFINED RK_BUILD_SHARED)
    set(RK_BUILD_SHARED ON)
endif()
option(MK_BUILD_SHARED "Build MotionKit as a shared library" ${RK_BUILD_SHARED})
if(NOT TARGET trajectory_core)
    set(TC_BUILD_SHARED ${MK_BUILD_SHARED} CACHE BOOL "Build the shared trajectory core" FORCE)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../trajectorykit/native" trajectory)
endif()

# Only Ruckig's offline Community calculator is linked. Its upstream defaults
# enable the networked cloud client, examples and tests, so override them here.
set(BUILD_CLOUD_CLIENT OFF)
set(BUILD_EXAMPLES OFF)
set(BUILD_PYTHON_MODULE OFF)
set(BUILD_TESTS OFF)
set(BUILD_BENCHMARK OFF)
set(BUILD_SHARED_LIBS OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ruckig/CMakeLists.txt")
    message(FATAL_ERROR "Initialize motionkit/native/vendor/ruckig submodule")
endif()
add_subdirectory(vendor/ruckig EXCLUDE_FROM_ALL)
set_target_properties(ruckig PROPERTIES POSITION_INDEPENDENT_CODE ON)
get_target_property(MK_RUCKIG_SOURCES ruckig SOURCES)
foreach(MK_RUCKIG_SOURCE IN LISTS MK_RUCKIG_SOURCES)
    if(MK_RUCKIG_SOURCE MATCHES "cloud_client")
        message(FATAL_ERROR "MotionKit must not compile Ruckig's cloud client")
    endif()
endforeach()

# Eigen is vendored (header-only, pinned to 3.4.0) so every platform, the browser included, builds against the same
# copy. A generated package config makes each find_package(Eigen3) in the build, TOPP-RA's too, resolve to it.
set(MK_EIGEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/eigen")
if(NOT EXISTS "${MK_EIGEN_DIR}/Eigen/Core")
    message(FATAL_ERROR "Initialize motionkit/native/vendor/eigen submodule")
endif()
set(MK_EIGEN_CONFIG_DIR "${CMAKE_CURRENT_BINARY_DIR}/eigen3")
file(WRITE "${MK_EIGEN_CONFIG_DIR}/Eigen3Config.cmake" "\
if(NOT TARGET Eigen3::Eigen)
    add_library(Eigen3::Eigen INTERFACE IMPORTED GLOBAL)
    set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES \"${MK_EIGEN_DIR}\")
endif()
set(EIGEN3_INCLUDE_DIR \"${MK_EIGEN_DIR}\")
set(EIGEN3_INCLUDE_DIRS \"${MK_EIGEN_DIR}\")
")
file(WRITE "${MK_EIGEN_CONFIG_DIR}/Eigen3ConfigVersion.cmake" "\
set(PACKAGE_VERSION 3.4.0)
if(PACKAGE_FIND_VERSION VERSION_GREATER PACKAGE_VERSION OR PACKAGE_FIND_VERSION_MAJOR GREATER 3)
    set(PACKAGE_VERSION_COMPATIBLE FALSE)
else()
    set(PACKAGE_VERSION_COMPATIBLE TRUE)
    if(PACKAGE_FIND_VERSION STREQUAL PACKAGE_VERSION)
        set(PACKAGE_VERSION_EXACT TRUE)
    endif()
endif()
")
set(Eigen3_DIR "${MK_EIGEN_CONFIG_DIR}" CACHE PATH "MotionKit's vendored Eigen" FORCE)

# TOPP-RA's Seidel solver is the only optimizer compiled for path timing.
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/toppra/cpp/CMakeLists.txt")
    message(FATAL_ERROR "Initialize motionkit/native/vendor/toppra submodule")
endif()
set(MK_OWN_BUILD_TESTING ${BUILD_TESTING})
set(BUILD_TESTING OFF)
set(PYTHON_BINDINGS OFF)
set(BUILD_WITH_PINOCCHIO OFF)
set(BUILD_WITH_qpOASES OFF)
set(BUILD_WITH_GLPK OFF)
add_subdirectory(vendor/toppra/cpp EXCLUDE_FROM_ALL)
# TOPP-RA maps the legacy BUILD_TESTS set above onto the BUILD_TESTING *cache* entry. Left switched off, the
# next configure of the same tree drops these tests while the first one's CTest files linger, so ctest
# lists them and reports "Not Run".
set(BUILD_TESTING ${MK_OWN_BUILD_TESTING} CACHE BOOL "Build tests" FORCE)
set(BUILD_TESTING ${MK_OWN_BUILD_TESTING})
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
get_target_property(MK_TOPPRA_SOURCES toppra SOURCES)
list(FILTER MK_TOPPRA_SOURCES EXCLUDE REGEX "(qpOASES|glpk|joint_torque|cartesian_velocity_norm|parametrizer)")
set_property(TARGET toppra PROPERTY SOURCES "${MK_TOPPRA_SOURCES}")
foreach(MK_TOPPRA_SOURCE IN LISTS MK_TOPPRA_SOURCES)
    if(MK_TOPPRA_SOURCE MATCHES "(qpOASES|glpk)")
        message(FATAL_ERROR "MotionKit must not compile TOPP-RA optional solvers")
    endif()
endforeach()

if(MK_BUILD_SHARED)
    set(MK_LIBRARY_TYPE SHARED)
else()
    set(MK_LIBRARY_TYPE STATIC)
endif()

# The accepted pinned C++ core is also the production 6R backend.
# Retain the evaluation executable as an optional, non-gating report.
option(MK_BUILD_EAIK_SPIKE "Build the pinned EAIK C++ parity spike" OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/eaik/CPP/external/ik-geo/cpp/subproblems/CMakeLists.txt")
    message(FATAL_ERROR "Initialize EAIK and its CPP/external/ik-geo submodule")
endif()
add_subdirectory(vendor/eaik/CPP/src eaik EXCLUDE_FROM_ALL)
foreach(MK_EAIK_TARGET EAIK EAIK_Kinematics SUBPROBLEMS)
    target_link_libraries(${MK_EAIK_TARGET} Eigen3::Eigen)
    set_target_properties(${MK_EAIK_TARGET} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(EMSCRIPTEN)
        target_compile_options(${MK_EAIK_TARGET} PRIVATE -fexceptions)
    endif()
endforeach()
find_package(Threads REQUIRED)
target_link_libraries(EAIK Threads::Threads)
if(MK_BUILD_EAIK_SPIKE)
    add_executable(motionkit_eaik_spike tests/eaik_spike.cpp)
    target_link_libraries(motionkit_eaik_spike PRIVATE motionkit_core EAIK Eigen3::Eigen)
    if(EMSCRIPTEN)
        target_compile_options(motionkit_eaik_spike PRIVATE -fexceptions)
        target_link_options(motionkit_eaik_spike PRIVATE -fexceptions)
    endif()
    # Compiled-model parity needs the exported fixture file; run it explicitly.
    # The standard native EAIK test supplies self-contained fixture coverage.
endif()

set(MK_DESCARTES_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/descartes_light/descartes_light/core/include")
if(NOT EXISTS "${MK_DESCARTES_INCLUDE}/descartes_light/solvers/ladder_graph/ladder_graph_solver.h")
    message(FATAL_ERROR "Initialize motionkit/native/vendor/descartes_light submodule")
endif()
# Emscripten's OpenMP needs threads, which the browser build does not use.
if(NOT EMSCRIPTEN)
    find_package(OpenMP QUIET COMPONENTS CXX)
endif()
add_library(motionkit_descartes_core STATIC src/descartes_instantiations.cpp)
set_target_properties(motionkit_descartes_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(motionkit_descartes_core PUBLIC cxx_std_17)
target_include_directories(motionkit_descartes_core PUBLIC "${MK_DESCARTES_INCLUDE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor/shims")
target_link_libraries(motionkit_descartes_core PUBLIC Eigen3::Eigen)
if(OpenMP_CXX_FOUND)
    target_link_libraries(motionkit_descartes_core PUBLIC OpenMP::OpenMP_CXX)
else()
    target_include_directories(motionkit_descartes_core PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/vendor/shims/no_openmp")
endif()

add_library(motionkit_core ${MK_LIBRARY_TYPE} src/generator.cpp src/eaik.cpp src/path.cpp src/analytic_cartesian.cpp src/orientation_lattice.cpp src/joint_lifts.cpp src/external_lattice.cpp src/candidate_sampling.cpp src/configuration.cpp src/structured_ladder.cpp src/path_differential.cpp src/redundancy_qp.cpp)
target_compile_features(motionkit_core PUBLIC cxx_std_17)
# Reuse the pinned ProxQP headers; no additional vendor or optional dependencies.
set(MK_PROXSUITE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../kinematicskit/native/vendor/proxsuite")
if(NOT EXISTS "${MK_PROXSUITE_DIR}/include/proxsuite/proxqp/sparse/sparse.hpp")
    message(FATAL_ERROR "Initialize kinematicskit/native/vendor/proxsuite submodule")
endif()
target_include_directories(motionkit_core SYSTEM PRIVATE
    "${MK_PROXSUITE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../kinematicskit/native/src/proxsuite_config")
target_link_libraries(motionkit_core PUBLIC trajectory_core PRIVATE EAIK ruckig::ruckig toppra Eigen3::Eigen motionkit_descartes_core)
target_include_directories(motionkit_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
if(EMSCRIPTEN)
    target_compile_options(motionkit_core PRIVATE -fexceptions)
    target_link_options(motionkit_core PUBLIC -fexceptions)
endif()
target_compile_definitions(motionkit_core PRIVATE MK_BUILDING_LIBRARY)
if(NOT MK_BUILD_SHARED)
    target_compile_definitions(motionkit_core PUBLIC MK_STATIC)
endif()
set_target_properties(motionkit_core PROPERTIES
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES
    VERSION ${PROJECT_VERSION}
    SOVERSION 0)

    add_test(NAME motionkit_path COMMAND motionkit_path_tests)

    # Every executable in this directory is a test; the library remains unaffected.
    get_property(motionkit_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(motionkit_target IN LISTS motionkit_targets)
        get_target_property(motionkit_type "${motionkit_target}" TYPE)
        if(motionkit_type STREQUAL "EXECUTABLE")

                target_compile_options("${motionkit_target}" PRIVATE -fexceptions)
                target_link_options("${motionkit_target}" PRIVATE -sSTACK_SIZE=1048576)
