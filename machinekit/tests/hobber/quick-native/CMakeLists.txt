cmake_minimum_required(VERSION 3.21)
project(hobber_quick_runtime LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(NOT DEFINED CACHED)
  message(FATAL_ERROR "Pass -DCACHED=<existing app native library directory>")
endif()
add_library(trajectory_core SHARED IMPORTED GLOBAL)
set_target_properties(trajectory_core PROPERTIES IMPORTED_LOCATION ${CACHED}/libtrajectory_core.so INTERFACE_INCLUDE_DIRECTORIES ${ROOT}/trajectorykit/native/include)
add_library(NativeKit::sim_core SHARED IMPORTED GLOBAL)
set_target_properties(NativeKit::sim_core PROPERTIES IMPORTED_LOCATION ${CACHED}/libnativekit_sim_core.so INTERFACE_INCLUDE_DIRECTORIES "${ROOT}/simkit/sim_core/include;${ROOT}/scenekit/scene/include;${ROOT}/haxeon/vendor/nativekit/include")
add_library(NativeKit::sim_mujoco SHARED IMPORTED GLOBAL)
set_target_properties(NativeKit::sim_mujoco PROPERTIES IMPORTED_LOCATION ${CACHED}/libnativekit_sim_mujoco.so INTERFACE_INCLUDE_DIRECTORIES ${ROOT}/simkit/sim_mujoco/include)
set(RK_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(RK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RK_BUILD_SIMKIT ON CACHE BOOL "" FORCE)
set(RK_BUILD_VIRTUAL_DEVICE OFF CACHE BOOL "" FORCE)
set(RK_BUILD_SERIAL_DEVICE OFF CACHE BOOL "" FORCE)
set(RK_BUILD_VISIONKIT OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/robotkit robotkit)
